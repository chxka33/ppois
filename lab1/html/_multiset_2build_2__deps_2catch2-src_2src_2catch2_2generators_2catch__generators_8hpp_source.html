89 template<typename T>
90 class FixedValuesGenerator final : public IGenerator<T> {
91 static_assert(!std::is_same<T, bool>::value,
92 "FixedValuesGenerator does not support bools because of std::vector<bool>"
93 "specialization, use SingleValue Generator instead.");
94 std::vector<T> m_values;
95 size_t m_idx = 0;
96 public:
97 FixedValuesGenerator( std::initializer_list<T> values ) : m_values( values ) {}
98
99 T const& get() const override {
100 return m_values[m_idx];
101 }
102 bool next() override {
103 ++m_idx;
104 return m_idx < m_values.size();
105 }

119 template<typename T>
120 class Generators : public IGenerator<T> {
121 std::vector<GeneratorWrapper<T>> m_generators;
122 size_t m_current = 0;
123
124 void add_generator( GeneratorWrapper<T>&& generator ) {
125 m_generators.emplace_back( CATCH_MOVE( generator ) );
126 }
127 void add_generator( T const& val ) {
128 m_generators.emplace_back( value( val ) );
129 }
130 void add_generator( T&& val ) {
131 m_generators.emplace_back( value( CATCH_MOVE( val ) ) );
132 }
133 template <typename U>
134 std::enable_if_t<!std::is_same<std::decay_t<U>, T>::value>
135 add_generator( U&& val ) {
136 add_generator( T( CATCH_FORWARD( val ) ) );
137 }
138
139 template <typename U> void add_generators( U&& valueOrGenerator ) {
140 add_generator( CATCH_FORWARD( valueOrGenerator ) );
141 }
142
143 template <typename U, typename... Gs>
144 void add_generators( U&& valueOrGenerator, Gs&&... moreGenerators ) {
145 add_generator( CATCH_FORWARD( valueOrGenerator ) );
146 add_generators( CATCH_FORWARD( moreGenerators )... );
147 }
148
149 public:
150 template <typename... Gs>
151 Generators(Gs &&... moreGenerators) {
152 m_generators.reserve(sizeof...(Gs));
153 add_generators(CATCH_FORWARD(moreGenerators)...);
154 }
155
156 T const& get() const override {
157 return m_generators[m_current].get();
158 }
159
160 bool next() override {
161 if (m_current >= m_generators.size()) {
162 return false;
163 }
164 const bool current_status = m_generators[m_current].next();
165 if (!current_status) {
166 ++m_current;
167 }
168 return m_current < m_generators.size();
169 }
170 };

206 template<typename L>
207 auto generate( StringRef generatorName, SourceLineInfo const& lineInfo, L const& generatorExpression ) -> typename decltype(generatorExpression())::type {
208 using UnderlyingType = typename decltype(generatorExpression())::type;
209
210 IGeneratorTracker* tracker = acquireGeneratorTracker( generatorName, lineInfo );
211 // Creation of tracker is delayed after generator creation, so
212 // that constructing generator can fail without breaking everything.
213 if (!tracker) {
214 tracker = createGeneratorTracker(
215 generatorName,
216 lineInfo,
217 Catch::Detail::make_unique<Generators<UnderlyingType>>(
218 generatorExpression() ) );
219 }
220
221 auto const& generator = static_cast<IGenerator<UnderlyingType> const&>( *tracker->getGenerator() );
222 return generator.get();
223 }
