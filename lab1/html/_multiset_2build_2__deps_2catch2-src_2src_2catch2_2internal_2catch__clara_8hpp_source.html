115 void loadBuffer();
116
117 public:
118 explicit TokenStream( Args const& args );

120
121 explicit operator bool() const {
122 return !m_tokenBuffer.empty() || it != itEnd;
123 }
124
125 size_t count() const {
126 return m_tokenBuffer.size() + ( itEnd - it );

166 template <typename T>
167 class ResultValueBase : public ResultBase {
168 public:
169 T const& value() const& {
170 enforceOk();
171 return m_value;
172 }
173 T&& value() && {
174 enforceOk();
175 return CATCH_MOVE( m_value );
176 }
177
178 protected:
179 ResultValueBase( ResultType type ): ResultBase( type ) {}
180
181 ResultValueBase( ResultValueBase const& other ):
182 ResultBase( other ) {
183 if ( m_type == ResultType::Ok )
184 new ( &m_value ) T( other.m_value );
185 }
186 ResultValueBase( ResultValueBase&& other ):
187 ResultBase( other ) {
188 if ( m_type == ResultType::Ok )
189 new ( &m_value ) T( CATCH_MOVE(other.m_value) );
190 }
191
192
193 ResultValueBase( ResultType, T const& value ):
194 ResultBase( ResultType::Ok ) {
195 new ( &m_value ) T( value );
196 }
197 ResultValueBase( ResultType, T&& value ):
198 ResultBase( ResultType::Ok ) {
199 new ( &m_value ) T( CATCH_MOVE(value) );
200 }
201
202 ResultValueBase& operator=( ResultValueBase const& other ) {
203 if ( m_type == ResultType::Ok )
204 m_value.~T();
205 ResultBase::operator=( other );
206 if ( m_type == ResultType::Ok )
207 new ( &m_value ) T( other.m_value );
208 return *this;
209 }
210 ResultValueBase& operator=( ResultValueBase&& other ) {
211 if ( m_type == ResultType::Ok ) m_value.~T();
212 ResultBase::operator=( other );
213 if ( m_type == ResultType::Ok )
214 new ( &m_value ) T( CATCH_MOVE(other.m_value) );
215 return *this;
216 }
217
218
219 ~ResultValueBase() override {
220 if ( m_type == ResultType::Ok )
221 m_value.~T();
222 }
223
224 union {
225 T m_value;
226 };
227 };

234 template <typename T = void>
235 class BasicResult : public ResultValueBase<T> {
236 public:
237 template <typename U>
238 explicit BasicResult( BasicResult<U> const& other ):
239 ResultValueBase<T>( other.type() ),
240 m_errorMessage( other.errorMessage() ) {
241 assert( type() != ResultType::Ok );
242 }
243
244 template <typename U>
245 static auto ok( U&& value ) -> BasicResult {
246 return { ResultType::Ok, CATCH_FORWARD(value) };
247 }
248 static auto ok() -> BasicResult { return { ResultType::Ok }; }
249 static auto logicError( std::string&& message )
250 -> BasicResult {
251 return { ResultType::LogicError, CATCH_MOVE(message) };
252 }
253 static auto runtimeError( std::string&& message )
254 -> BasicResult {
255 return { ResultType::RuntimeError, CATCH_MOVE(message) };
256 }
257
258 explicit operator bool() const {
259 return m_type == ResultType::Ok;
260 }
261 auto type() const -> ResultType { return m_type; }
262 auto errorMessage() const -> std::string const& {
263 return m_errorMessage;
264 }
265
266 protected:
267 void enforceOk() const override {
268
269 // Errors shouldn't reach this point, but if they do
270 // the actual error message will be in m_errorMessage
271 assert( m_type != ResultType::LogicError );
272 assert( m_type != ResultType::RuntimeError );
273 if ( m_type != ResultType::Ok )
274 std::abort();
275 }
276
277 std::string
278 m_errorMessage; // Only populated if resultType is an error
279
280 BasicResult( ResultType type,
281 std::string&& message ):
282 ResultValueBase<T>( type ), m_errorMessage( CATCH_MOVE(message) ) {
283 assert( m_type != ResultType::Ok );
284 }
285
286 using ResultValueBase<T>::ResultValueBase;
287 using ResultBase::m_type;
288 };

317 template <typename T>
318 ParserResult convertInto( std::string const& source, T& target ) {
319 std::stringstream ss( source );
320 ss >> target;
321 if ( ss.fail() ) {
322 return ParserResult::runtimeError(
323 "Unable to convert '" + source +
324 "' to destination type" );
325 } else {
326 return ParserResult::ok( ParseResultType::Matched );
327 }
328 }
329 ParserResult convertInto( std::string const& source,
330 std::string& target );
331 ParserResult convertInto( std::string const& source, bool& target );
332
333#ifdef CLARA_CONFIG_OPTIONAL_TYPE
334 template <typename T>
335 auto convertInto( std::string const& source,
336 CLARA_CONFIG_OPTIONAL_TYPE<T>& target )
337 -> ParserResult {
338 T temp;
339 auto result = convertInto( source, temp );
340 if ( result )
341 target = CATCH_MOVE( temp );
342 return result;
343 }
344#endif // CLARA_CONFIG_OPTIONAL_TYPE

396 template <typename ReturnType> struct LambdaInvoker {
397 static_assert(
398 std::is_same<ReturnType, ParserResult>::value,
399 "Lambda must return void or clara::ParserResult" );
400
401 template <typename L, typename ArgType>
402 static auto invoke( L const& lambda, ArgType const& arg )
403 -> ParserResult {
404 return lambda( arg );
405 }
406 };
407
408 template <> struct LambdaInvoker<void> {
409 template <typename L, typename ArgType>
410 static auto invoke( L const& lambda, ArgType const& arg )
411 -> ParserResult {
412 lambda( arg );
413 return ParserResult::ok( ParseResultType::Matched );
414 }
415 };
416
417 template <typename ArgType, typename L>
418 auto invokeLambda( L const& lambda, std::string const& arg )
419 -> ParserResult {
420 ArgType temp{};
421 auto result = convertInto( arg, temp );
422 return !result ? result
423 : LambdaInvoker<typename UnaryLambdaTraits<
424 L>::ReturnType>::invoke( lambda, temp );
425 }

427 template <typename L> struct BoundLambda : BoundValueRefBase {
428 L m_lambda;
429
430 static_assert(
431 UnaryLambdaTraits<L>::isValid,
432 "Supplied lambda must take exactly one argument" );
433 explicit BoundLambda( L const& lambda ): m_lambda( lambda ) {}
434
435 auto setValue( std::string const& arg )
436 -> ParserResult override {
437 return invokeLambda<typename UnaryLambdaTraits<L>::ArgType>(
438 m_lambda, arg );
439 }

541 auto isOptional() const -> bool {
