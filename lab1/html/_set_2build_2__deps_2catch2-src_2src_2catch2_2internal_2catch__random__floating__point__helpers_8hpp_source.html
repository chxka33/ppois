28 template <typename FloatType>
29 FloatType gamma(FloatType a, FloatType b) {
30 static_assert( std::is_floating_point<FloatType>::value,
31 "gamma returns the largest ULP magnitude within "
32 "floating point range [a, b]. This only makes sense "
33 "for floating point types" );
34 assert( a <= b );
35
36 const auto gamma_up = Catch::nextafter( a, std::numeric_limits<FloatType>::infinity() ) - a;
37 const auto gamma_down = b - Catch::nextafter( b, -std::numeric_limits<FloatType>::infinity() );
38
39 return gamma_up < gamma_down ? gamma_down : gamma_up;
40 }

69 template <typename FloatType>
70 DistanceType<FloatType>
71 count_equidistant_floats( FloatType a, FloatType b, FloatType distance ) {
72 assert( a <= b );
73 // We get distance as gamma for our uniform float distribution,
74 // so this will round perfectly.
75 const auto ag = a / distance;
76 const auto bg = b / distance;
77
78 const auto s = bg - ag;
79 const auto err = ( std::fabs( a ) <= std::fabs( b ) )
80 ? -ag - ( s - bg )
81 : bg - ( s + ag );
82 const auto ceil_s = static_cast<DistanceType<FloatType>>( std::ceil( s ) );
83
84 return ( ceil_s != s ) ? ceil_s : ceil_s + ( err > 0 );
85 }
