43template <typename IntegerType>
44class uniform_integer_distribution {
45 static_assert(std::is_integral<IntegerType>::value, "...");
46
47 using UnsignedIntegerType = Detail::make_unsigned_t<IntegerType>;
48
49 // Only the left bound is stored, and we store it converted to its
50 // unsigned image. This avoids having to do the conversions inside
51 // the operator(), at the cost of having to do the conversion in
52 // the a() getter. The right bound is only needed in the b() getter,
53 // so we recompute it there from other stored data.
54 UnsignedIntegerType m_a;
55
56 // How many different values are there in [a, b]. a == b => 1, can be 0 for distribution over all values in the type.
57 UnsignedIntegerType m_ab_distance;
58
59 // We hoisted this out of the main generation function. Technically,
60 // this means that using this distribution will be slower than Lemire's
61 // algorithm if this distribution instance will be used only few times,
62 // but it will be faster if it is used many times. Since Catch2 uses
63 // distributions only to implement random generators, we assume that each
64 // distribution will be reused many times and this is an optimization.
65 UnsignedIntegerType m_rejection_threshold = 0;
66
67 UnsignedIntegerType computeDistance(IntegerType a, IntegerType b) const {
68 // This overflows and returns 0 if a == 0 and b == TYPE_MAX.
69 // We handle that later when generating the number.
70 return transposeTo(b) - transposeTo(a) + 1;
71 }
72
73 static UnsignedIntegerType computeRejectionThreshold(UnsignedIntegerType ab_distance) {
74 // distance == 0 means that we will return all possible values from
75 // the type's range, and that we shouldn't reject anything.
76 if ( ab_distance == 0 ) { return 0; }
77 return ( ~ab_distance + 1 ) % ab_distance;
78 }
79
80 static UnsignedIntegerType transposeTo(IntegerType in) {
81 return Detail::transposeToNaturalOrder<IntegerType>(
82 static_cast<UnsignedIntegerType>( in ) );
83 }
84 static IntegerType transposeBack(UnsignedIntegerType in) {
85 return static_cast<IntegerType>(
86 Detail::transposeToNaturalOrder<IntegerType>(in) );
87 }
88
89public:
90 using result_type = IntegerType;
91
92 uniform_integer_distribution( IntegerType a, IntegerType b ):
93 m_a( transposeTo(a) ),
94 m_ab_distance( computeDistance(a, b) ),
95 m_rejection_threshold( computeRejectionThreshold(m_ab_distance) ) {
96 assert( a <= b );
97 }
98
99 template <typename Generator>
100 result_type operator()( Generator& g ) {
101 // All possible values of result_type are valid.
102 if ( m_ab_distance == 0 ) {
103 return transposeBack( Detail::fillBitsFrom<UnsignedIntegerType>( g ) );
104 }
105
106 auto random_number = Detail::fillBitsFrom<UnsignedIntegerType>( g );
107 auto emul = Detail::extendedMult( random_number, m_ab_distance );
108 // Unlike Lemire's algorithm we skip the ab_distance check, since
109 // we precomputed the rejection threshold, which is always tighter.
110 while (emul.lower < m_rejection_threshold) {
111 random_number = Detail::fillBitsFrom<UnsignedIntegerType>( g );
112 emul = Detail::extendedMult( random_number, m_ab_distance );
113 }
114
115 return transposeBack(m_a + emul.upper);
116 }
117
118 result_type a() const { return transposeBack(m_a); }
119 result_type b() const { return transposeBack(m_ab_distance + m_a - 1); }
120};
